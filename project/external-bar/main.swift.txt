// The external bar from ADR-001: the same bytes, judged by both implementations.
//
// Not a port of their tests. Each case builds one ceremony and hands the identical bytes to
// SwiftWebAuthn and to brokenhandsio/swift-webauthn, then compares accept against accept. A
// disagreement is a finding in one of them, and which one is a question the specification
// answers rather than a vote.
//
// The key material is theirs — the P-256 pair published in their test suite — so the vectors are
// not ours either.

import Crypto
import Foundation
import SwiftWebAuthn
import WebAuthn

// MARK: - Their published key

let theirPrivateKeyPEM = """
-----BEGIN PRIVATE KEY-----
MIGHAgEAMBMGByqGSM49AgEGCCqGSM49AwEHBG0wawIBAQQgUC6oOLmd9F3Ak32L
WJCzQB1eF00UX5MCzYi47hNS+zqhRANCAASWIbQJIqS1L1E8G2Z5uNSPgQGZcsfz
xk1shW3jTkWmRWY3MSr+CumivsCLz0YR4OkIHm8SAxGomGYF1dO0skj4
-----END PRIVATE KEY-----
"""

let privateKey = try P256.Signing.PrivateKey(pemRepresentation: theirPrivateKeyPEM)
let x963 = Array(privateKey.publicKey.x963Representation)
let coordinateX = Array(x963[1..<33])
let coordinateY = Array(x963[33..<65])

let relyingPartyID = "example.com"
let origin = "https://example.com"
let credentialID = [UInt8](repeating: 0xC1, count: 32)

// MARK: - Byte builders

func base64url(_ bytes: [UInt8]) -> String {
    Data(bytes).base64EncodedString()
        .replacingOccurrences(of: "+", with: "-")
        .replacingOccurrences(of: "/", with: "_")
        .replacingOccurrences(of: "=", with: "")
}

func byteString(_ bytes: [UInt8]) -> [UInt8] {
    if bytes.count < 24 { return [0x40 | UInt8(bytes.count)] + bytes }
    if bytes.count <= 255 { return [0x58, UInt8(bytes.count)] + bytes }
    return [0x59, UInt8(bytes.count >> 8), UInt8(bytes.count & 0xFF)] + bytes
}

func textString(_ text: String) -> [UInt8] {
    let bytes = Array(text.utf8)
    return [0x60 | UInt8(bytes.count)] + bytes
}

/// `{1: 2, 3: -7, -1: 1, -2: x, -3: y}`
let coseKey: [UInt8] = [0xA5]
    + [0x01, 0x02] + [0x03, 0x26] + [0x20, 0x01]
    + [0x21] + byteString(coordinateX)
    + [0x22] + byteString(coordinateY)

struct Flags {
    static let userPresent: UInt8 = 0x01
    static let userVerified: UInt8 = 0x04
    static let backupEligible: UInt8 = 0x08
    static let backedUp: UInt8 = 0x10
    static let attested: UInt8 = 0x40
}

func authenticatorData(
    rpID: String = relyingPartyID,
    flags: UInt8,
    counter: UInt32 = 0,
    attested: Bool,
    credentialIDLengthOverride: UInt16? = nil
) -> [UInt8] {
    var bytes = Array(SHA256.hash(data: Data(rpID.utf8)))
    bytes.append(flags)
    bytes += [
        UInt8((counter >> 24) & 0xFF), UInt8((counter >> 16) & 0xFF),
        UInt8((counter >> 8) & 0xFF), UInt8(counter & 0xFF),
    ]
    if attested {
        bytes += [UInt8](repeating: 0, count: 16)
        let declared = credentialIDLengthOverride ?? UInt16(credentialID.count)
        bytes += [UInt8(declared >> 8), UInt8(declared & 0xFF)]
        bytes += credentialID
        bytes += coseKey
    }
    return bytes
}

func clientData(type: String, challenge: [UInt8], origin: String = origin) -> [UInt8] {
    Array("""
        {"type":"\(type)","challenge":"\(base64url(challenge))","origin":"\(origin)","crossOrigin":false}
        """.utf8)
}

func attestationObject(format: String = "none", statement: [UInt8] = [0xA0], authData: [UInt8]) -> [UInt8] {
    [0xA3]
        + textString("fmt") + textString(format)
        + textString("attStmt") + statement
        + textString("authData") + byteString(authData)
}

// MARK: - Running both

enum Verdict: Equatable {
    case accepted
    case rejected(String)

    var label: String {
        switch self {
        case .accepted: "accept"
        case .rejected: "reject"
        }
    }
}

let ours = SwiftWebAuthn.RelyingParty(
    id: relyingPartyID, name: "External bar", origins: [origin]
)
let theirs = WebAuthnManager(configuration: .init(
    relyingPartyID: relyingPartyID,
    relyingPartyName: "External bar",
    relyingPartyOrigin: origin
))

func ourRegistration(clientDataJSON: [UInt8], attestation: [UInt8], challenge: [UInt8], requireUV: Bool) -> Verdict {
    do {
        _ = try ours.verifyRegistration(
            SwiftWebAuthn.RegistrationResponse(
                clientDataJSON: clientDataJSON, attestationObject: attestation
            ),
            expecting: SwiftWebAuthn.Challenge(challenge),
            requiringUserVerification: requireUV
        )
        return .accepted
    } catch {
        return .rejected("\(error)")
    }
}

func theirRegistration(clientDataJSON: [UInt8], attestation: [UInt8], challenge: [UInt8], requireUV: Bool) async -> Verdict {
    let json = """
        {"id":"\(base64url(credentialID))","type":"public-key","rawId":"\(base64url(credentialID))",
         "response":{"clientDataJSON":"\(base64url(clientDataJSON))",
                     "attestationObject":"\(base64url(attestation))"}}
        """
    do {
        let credential = try JSONDecoder().decode(RegistrationCredential.self, from: Data(json.utf8))
        _ = try await theirs.finishRegistration(
            challenge: challenge,
            credentialCreationData: credential,
            requireUserVerification: requireUV,
            confirmCredentialIDNotRegisteredYet: { _ in true }
        )
        return .accepted
    } catch {
        return .rejected("\(error)")
    }
}

func ourAuthentication(
    clientDataJSON: [UInt8], authData: [UInt8], signature: [UInt8],
    challenge: [UInt8], storedCounter: UInt32, storedBackedUp: Bool, requireUV: Bool
) -> Verdict {
    do {
        let stored = SwiftWebAuthn.RegisteredCredential(
            id: credentialID,
            publicKey: try SwiftWebAuthn.COSEPublicKey(coseKey),
            signCount: storedCounter,
            isBackupEligible: storedBackedUp,
            isBackedUp: storedBackedUp,
            aaguid: [UInt8](repeating: 0, count: 16),
            attestationFormat: "none"
        )
        _ = try ours.verifyAuthentication(
            SwiftWebAuthn.AuthenticationResponse(
                credentialID: credentialID, clientDataJSON: clientDataJSON,
                authenticatorData: authData, signature: signature, userHandle: nil
            ),
            expecting: SwiftWebAuthn.Challenge(challenge),
            against: stored,
            requiringUserVerification: requireUV
        )
        return .accepted
    } catch {
        return .rejected("\(error)")
    }
}

func theirAuthentication(
    clientDataJSON: [UInt8], authData: [UInt8], signature: [UInt8],
    challenge: [UInt8], storedCounter: UInt32, requireUV: Bool
) -> Verdict {
    let json = """
        {"id":"\(base64url(credentialID))","type":"public-key","rawId":"\(base64url(credentialID))",
         "response":{"clientDataJSON":"\(base64url(clientDataJSON))",
                     "authenticatorData":"\(base64url(authData))",
                     "signature":"\(base64url(signature))"}}
        """
    do {
        let credential = try JSONDecoder().decode(AuthenticationCredential.self, from: Data(json.utf8))
        _ = try theirs.finishAuthentication(
            credential: credential,
            expectedChallenge: challenge,
            credentialPublicKey: coseKey,
            credentialCurrentSignCount: storedCounter,
            requireUserVerification: requireUV
        )
        return .accepted
    } catch {
        return .rejected("\(error)")
    }
}

// MARK: - The cases

var agreements = 0
var disagreements: [(String, Verdict, Verdict)] = []

/// One case, in a form any language can read.
struct CorpusCase: Encodable {
    let name: String
    let ceremony: String
    let relyingPartyID: String
    let origin: String
    let challenge: String
    let requireUserVerification: Bool
    let credentialID: String
    var clientDataJSON: String
    var attestationObject: String?
    var authenticatorData: String?
    var signature: String?
    var coseKey: String?
    var storedCounter: UInt32?
    var storedBackedUp: Bool?
    var ours: String
    var theirs: String
}

var corpus: [CorpusCase] = []

/// Records a case for the other implementations, alongside the two verdicts reached here.
@MainActor func record(
    _ name: String, ceremony: String, mine: Verdict, other: Verdict,
    clientDataJSON: [UInt8], challenge: [UInt8], requireUV: Bool,
    rpID: String = relyingPartyID, origin originValue: String = origin,
    credential: [UInt8] = credentialID,
    attestation: [UInt8]? = nil, authData: [UInt8]? = nil, signature: [UInt8]? = nil,
    cose: [UInt8]? = nil, storedCounter: UInt32? = nil, storedBackedUp: Bool? = nil
) {
    corpus.append(CorpusCase(
        name: name, ceremony: ceremony, relyingPartyID: rpID, origin: originValue,
        challenge: base64url(challenge), requireUserVerification: requireUV,
        credentialID: base64url(credential),
        clientDataJSON: base64url(clientDataJSON),
        attestationObject: attestation.map(base64url),
        authenticatorData: authData.map(base64url),
        signature: signature.map(base64url),
        coseKey: cose.map(base64url),
        storedCounter: storedCounter, storedBackedUp: storedBackedUp,
        ours: mine.label, theirs: other.label
    ))
}

@MainActor func compare(_ name: String, _ mine: Verdict, _ theirs: Verdict) {
    let same = mine.label == theirs.label
    if same {
        agreements += 1
        print("  ✓ \(name.padding(toLength: 52, withPad: " ", startingAt: 0)) both \(mine.label)")
    } else {
        disagreements.append((name, mine, theirs))
        print("  ✗ \(name.padding(toLength: 52, withPad: " ", startingAt: 0)) ours \(mine.label), theirs \(theirs.label)")
    }
}


// MARK: - The corpus

let challenge = [UInt8](repeating: 0x5A, count: 32)
let authChallenge = [UInt8](repeating: 0x7B, count: 32)

let validAuthData = authenticatorData(
    flags: Flags.userPresent | Flags.userVerified | Flags.attested, attested: true
)
let validClient = clientData(type: "webauthn.create", challenge: challenge)
let validAttestation = attestationObject(authData: validAuthData)
let assertionClient = clientData(type: "webauthn.get", challenge: authChallenge)

/// Runs one registration payload through both, and records it for the others.
@MainActor func checkRegistration(
    _ name: String,
    clientDataJSON: [UInt8] = validClient,
    attestation: [UInt8],
    expecting expected: [UInt8] = challenge,
    requireUV: Bool = true
) async {
    let mine = ourRegistration(clientDataJSON: clientDataJSON, attestation: attestation,
                               challenge: expected, requireUV: requireUV)
    let other = await theirRegistration(clientDataJSON: clientDataJSON, attestation: attestation,
                                        challenge: expected, requireUV: requireUV)
    compare(name, mine, other)
    record(name, ceremony: "registration", mine: mine, other: other,
           clientDataJSON: clientDataJSON, challenge: expected, requireUV: requireUV,
           attestation: attestation)
}

/// Runs one assertion through both, signing it with the published key, and records it.
@MainActor func checkAuthentication(
    _ name: String, reported: UInt32, stored: UInt32, backedUp: Bool,
    corrupt: Bool = false, foreignKey: Bool = false
) throws {
    var flags = Flags.userPresent | Flags.userVerified
    if backedUp { flags |= Flags.backupEligible | Flags.backedUp }
    let data = authenticatorData(flags: flags, counter: reported, attested: false)
    let hash = Array(SHA256.hash(data: Data(assertionClient)))
    let signer = foreignKey ? P256.Signing.PrivateKey() : privateKey
    var signature = Array(try signer.signature(for: Data(data + hash)).derRepresentation)
    if corrupt { signature[signature.count - 1] ^= 0xFF }

    let mine = ourAuthentication(clientDataJSON: assertionClient, authData: data, signature: signature,
                                 challenge: authChallenge, storedCounter: stored,
                                 storedBackedUp: backedUp, requireUV: true)
    let other = theirAuthentication(clientDataJSON: assertionClient, authData: data, signature: signature,
                                    challenge: authChallenge, storedCounter: stored, requireUV: true)
    compare(name, mine, other)
    record(name, ceremony: "authentication", mine: mine, other: other,
           clientDataJSON: assertionClient, challenge: authChallenge, requireUV: true,
           authData: data, signature: signature, cose: coseKey,
           storedCounter: stored, storedBackedUp: backedUp)
}

print("\nREGISTRATION\n")

await checkRegistration("a valid none-attestation ceremony", attestation: validAttestation)
await checkRegistration("a challenge nobody issued", attestation: validAttestation,
                        expecting: [UInt8](repeating: 0x11, count: 32))
await checkRegistration("a look-alike origin",
                        clientDataJSON: clientData(type: "webauthn.create", challenge: challenge,
                                                   origin: "https://example.com.evil.test"),
                        attestation: validAttestation)
await checkRegistration("another relying party's hash",
                        attestation: attestationObject(authData: authenticatorData(
                            rpID: "other.example",
                            flags: Flags.userPresent | Flags.userVerified | Flags.attested,
                            attested: true)))
await checkRegistration("the authentication ceremony's client data",
                        clientDataJSON: clientData(type: "webauthn.get", challenge: challenge),
                        attestation: validAttestation)
await checkRegistration("nobody present",
                        attestation: attestationObject(authData: authenticatorData(
                            flags: Flags.attested, attested: true)))
await checkRegistration("present but unverified, verification required",
                        attestation: attestationObject(authData: authenticatorData(
                            flags: Flags.userPresent | Flags.attested, attested: true)))
await checkRegistration("a credential-id length beyond the data",
                        attestation: attestationObject(authData: authenticatorData(
                            flags: Flags.userPresent | Flags.userVerified | Flags.attested,
                            attested: true, credentialIDLengthOverride: 4096)))
await checkRegistration("an attestation format that cannot be checked",
                        attestation: attestationObject(format: "tpm", statement: [0xA0],
                                                       authData: validAuthData))

let duplicateKey: [UInt8] = [0xA4]
    + textString("fmt") + textString("none")
    + textString("fmt") + textString("packed")
    + textString("attStmt") + [0xA0]
    + textString("authData") + byteString(validAuthData)
await checkRegistration("a duplicate CBOR map key", attestation: duplicateKey)

print("\nCBOR MALLEABILITY\n")

await checkRegistration("one trailing zero byte", attestation: validAttestation + [0x00])
await checkRegistration("a second complete CBOR item appended",
                        attestation: validAttestation + [0xA1, 0x61, 0x78, 0x01])
await checkRegistration("1024 bytes of padding appended",
                        attestation: validAttestation + [UInt8](repeating: 0xFF, count: 1024))

let extraKey: [UInt8] = [0xA4]
    + textString("fmt") + textString("none")
    + textString("attStmt") + [0xA0]
    + textString("authData") + byteString(validAuthData)
    + textString("extra") + textString("ignored")
await checkRegistration("an unknown fourth key inside the map", attestation: extraKey)

let nonCanonicalLength: [UInt8] = [0xA3]
    + textString("fmt") + [0x78, 0x04] + Array("none".utf8)
    + textString("attStmt") + [0xA0]
    + textString("authData") + byteString(validAuthData)
await checkRegistration("a non-canonical string length", attestation: nonCanonicalLength)

var indefinite: [UInt8] = [0xBF]
indefinite += textString("fmt") + textString("none")
indefinite += textString("attStmt") + [0xA0]
indefinite += textString("authData") + byteString(validAuthData)
indefinite += [0xFF]
await checkRegistration("an indefinite-length map", attestation: indefinite)

print("\nAUTHENTICATION\n")

try checkAuthentication("a valid assertion", reported: 5, stored: 4, backedUp: false)
try checkAuthentication("a signature over different bytes", reported: 5, stored: 4, backedUp: false, corrupt: true)
try checkAuthentication("a signature by a key nobody registered", reported: 5, stored: 4, backedUp: false, foreignKey: true)

print("\nCOUNTER POLICY\n")

try checkAuthentication("0 → 0, not backed up  (counter unsupported)", reported: 0, stored: 0, backedUp: false)
try checkAuthentication("0 → 0, backed up      (the Apple case)", reported: 0, stored: 0, backedUp: true)
try checkAuthentication("8 → 9, not backed up  (normal advance)", reported: 9, stored: 8, backedUp: false)
try checkAuthentication("8 → 9, backed up", reported: 9, stored: 8, backedUp: true)
try checkAuthentication("8 → 8, not backed up  (stalled)", reported: 8, stored: 8, backedUp: false)
try checkAuthentication("8 → 8, backed up", reported: 8, stored: 8, backedUp: true)
try checkAuthentication("8 → 3, not backed up  (regressed)", reported: 3, stored: 8, backedUp: false)
try checkAuthentication("8 → 3, backed up", reported: 3, stored: 8, backedUp: true)
try checkAuthentication("8 → 0, backed up", reported: 0, stored: 8, backedUp: true)
try checkAuthentication("0 → 5, not backed up  (first advance)", reported: 5, stored: 0, backedUp: false)

// MARK: - Emit the corpus for other languages

let encoder = JSONEncoder()
encoder.outputFormatting = [.prettyPrinted, .sortedKeys]
let corpusURL = URL(fileURLWithPath: FileManager.default.currentDirectoryPath)
    .appendingPathComponent("corpus.json")
try encoder.encode(corpus).write(to: corpusURL)

print("""

    \(agreements) agreements, \(disagreements.count) disagreements
    corpus: \(corpus.count) cases → \(corpusURL.path)
    """)

for (name, mine, other) in disagreements {
    print("  · \(name): ours \(mine.label), theirs \(other.label)")
}
